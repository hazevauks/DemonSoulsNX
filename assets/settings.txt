# Demon's Souls for Nintendo Switch: options for the game (demonsouls_game.nro).
#
# One NAME=value a line, read as the game starts; a line starting with # is left out, so take
# the # off a line to use it. These are the options ps3recomp's runtime and Soulbrandt read from
# environment variables on a PC. The log lists the ones it set ("settings:").

# Skip the boot logos (2: shorten them).
#DES_SKIP_INTRO=1

# Skip the opening movie.
#DES_SKIP_INTRO_MOVIE=1

# Swap buttons, the game's = the controller's PS3 name. The default is by position:
# B is Cross, A Circle, Y Square, X Triangle. For confirming with A:
#PAD_MAP=cross=circle,circle=cross

# The runtime's per-event lines in the log (1), or none (0). More lines, a slower game.
#PS3_VERBOSE=1

# The memory the game's cache (/dev_hdd1) may use, in MB; files past it go to the SD card.
# 0 keeps the whole cache on the SD card, as before (slower). The default is 512.
#DESNX_CACHE_MEMORY_MB=512

# The renderer: deko3d (the default) or null (checks every command, draws nothing).
#RSX_BACKEND=null

# deko3D tests (docs/deko3d.md): the first N GPU commands one by one, naming one that faults;
# and black clears drawn as a slowly changing colour, to see the frames reach the screen.
#DESNX_DK_TRACE=4000
#DESNX_DK_TINT=1

# deko3D's targets: new ones start as zeroes (0 leaves them as their memory was); the hardware's
# compression on them (0 off); each kind of target, copy and view named in the log once.
#DESNX_DK_CLEAR_NEW=0
#DESNX_DK_COMPRESS=0
#DESNX_DK_CENSUS=1
# With 1, L3 and R3 together write the frame's colour targets to SD_LOG/rt_dump as pictures (the
# game stops for about 30 seconds; the pictures are the game's, keep them off the repository).
#DESNX_DK_RT_DUMP=1
# The first draw of each pipeline into a target smaller than the screen, described in the log.
#DESNX_DK_DRAW_PROBE=1
# 0 keeps the vertex fetch's component loop unrolled (slower compiles; the same pictures so far).
#DESNX_DK_FETCH_LOOP=0

# The game's shaders, compiled on the console: each step's time in the log, and the GLSL of one
# that fails. Soulbrandt's pass over every shader at the start is off here; 1 runs it.
#DESNX_DK_SHADER_TIMES=1
#DESNX_DK_SHADER_DUMP=1
# Every program's HLSL to shader_cache/hlsl on the SD card (made from the game: keep it off the repository).
#DESNX_DK_HLSL_DUMP=1
#DES_PRECOMPILE=1

# Each port's reads of the pad, in the log.
#PAD_TRACE=1

# Crash on purpose this many seconds in, to try the crash report (SD_LOG/demonsouls_crash.txt).
#DESNX_CRASH_TEST=30
